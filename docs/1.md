ÉTAPE 2 — Mapper XML

Créer :

src/services/prestaMapper.js
CODE
export function xmlToEmployees(xmlString) {

  const xml =
    new DOMParser()
      .parseFromString(
        xmlString,
        "text/xml"
      );



  return [

    ...xml.getElementsByTagName(
      "employee"
    )

  ].map((e) => {

    return {

      id:
        e.querySelector("id")
          ?.textContent
          ?.trim(),



      firstname:
        e.querySelector("firstname")
          ?.textContent
          ?.trim(),



      lastname:
        e.querySelector("lastname")
          ?.textContent
          ?.trim(),



      email:
        e.querySelector("email")
          ?.textContent
          ?.trim()

    };

  });

}

<script setup>

import {

  ref,
  onMounted,
  computed

}

from "vue";



import api
from "../../services/prestashopAPI";



import {
  xmlToEmployees
}

from "../../services/prestaMapper";



import EmployeeCard
from "../../components/employee/EmployeeCard.vue";



const employees =
  ref([]);



// computed
const totalEmployees =
  computed(() => {

    return employees.value.length;

});



async function loadEmployees() {

  try {

    const res =
      await api.get(
        "/employees?display=full"
      );



    employees.value =
      xmlToEmployees(
        res.data
      );



    console.log(
      employees.value
    );

  }

  catch (error) {

    console.log(error);

  }

}



// lifecycle
onMounted(() => {

  loadEmployees();

});

</script>





<template>

  <div>

    <h1>
      Employees
    </h1>



    <p>

      Nombre :
      {{ totalEmployees }}

    </p>



    <EmployeeCard

      v-for="e in employees"

      :key="e.id"

      :employee="e"

    />

  </div>

</template>
